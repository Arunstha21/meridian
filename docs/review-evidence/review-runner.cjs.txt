const fs = require('node:fs');
const crypto = require('node:crypto');
const { spawnSync } = require('node:child_process');
const { Client } = require('pg');
const dotenv = require('dotenv');

(async () => {
  const base = new URL('postgres://review_admin@127.0.0.1:55439/postgres');
  if (!['localhost', '127.0.0.1', '::1'].includes(base.hostname)) throw new Error('Local database required');
  const name = 'meridian_review_' + crypto.randomBytes(8).toString('hex');
  const adminUrl = new URL(base); adminUrl.pathname = '/postgres';
  const admin = new Client({ connectionString: adminUrl.href });
  await admin.connect();
  const exists = await admin.query('SELECT 1 FROM pg_database WHERE datname = $1', [name]);
  if (exists.rowCount) throw new Error('Refusing to use an existing database');
  base.pathname = '/' + name;
  const env = { ...process.env, DATABASE_URL: base.href, TEST_DATABASE_URL: base.href,
    NODE_ENV: 'test', LOG_LEVEL: 'error', MAIL_TRANSPORT: 'console',
    ADMIN_EMAILS: 'review-platform-admin@example.test', APP_URL: 'http://localhost:3000' };
  console.log('Using new isolated database: ' + name);
  try {
    if (process.argv.includes('--probes-only')) {
      await admin.query('CREATE DATABASE "' + name + '"');
      const migration = spawnSync(process.execPath, ['--import', 'tsx', 'bin/migrate.ts', 'up'], { env, encoding: 'utf8' });
      if (migration.status !== 0) throw new Error('Review migration failed: ' + migration.stderr);
    } else {
    const test = spawnSync(process.execPath, ['node_modules/vitest/vitest.mjs', 'run'], { env, encoding: 'utf8' });
    fs.writeFileSync('.playwright-mcp/review-tests.log', (test.stdout || '') + (test.stderr || ''));
    console.log('Existing suite exit: ' + test.status);
    console.log((test.stdout || '').slice(-4500));
    console.log((test.stderr || '').slice(-2500));
    }
    const probes = spawnSync(process.execPath, ['--import', 'tsx', '.playwright-mcp/review-probes.ts'], { env, encoding: 'utf8' });
    fs.writeFileSync('.playwright-mcp/review-probes.log', (probes.stdout || '') + (probes.stderr || ''));
    console.log(probes.stdout || ''); console.log(probes.stderr || '');
    console.log('Review probes exit: ' + probes.status);
  } finally {
    if (!/^meridian_review_[a-f0-9]{16}$/.test(name) || base.pathname !== '/' + name) throw new Error('Cleanup guard failed');
    await admin.query('DROP DATABASE IF EXISTS "' + name + '" WITH (FORCE)');
    await admin.end();
    console.log('Removed isolated review database.');
  }
})().catch(e => { console.error(e.message); process.exitCode = 1; });
